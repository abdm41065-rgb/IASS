    const code='ACT'+Date.now().toString(36).toUpperCase().slice(-6);
    const sections=[...document.querySelectorAll('.actSecCheck:checked')].map(cb=>cb.value);
    const newCode={id:'ac'+Date.now(),code,service:'accounting',plan:planKey,sections,durationDays:30,maxUses:1,uses:0,active:true};
    activationCodes.unshift(newCode);saveActivationCodes();

  }

  if(action==='delete'){if(!confirm('حذف هذا الطلب؟'))return;subRequests=subRequests.filter(r=>r.id!==id);saveRequests(subRequests);renderAll();showNotif('تم الحذف','info');}

async function fbInit(){
  try {
    const [fbMembers, fbCodes, fbActCodes, fbReqs] = await Promise.all([
      dbLoad(STORAGE_KEY), dbLoad(DISCOUNT_KEY), dbLoad(ACTIVATION_KEY), dbLoad(REQUESTS_KEY)
    ]);
    if(fbMembers.length) members = fbMembers.map(normMember);
    if(fbCodes.length) discountCodes = fbCodes;
    if(fbActCodes.length) activationCodes = fbActCodes;
    if(fbReqs.length) subRequests = fbReqs;
    renderAll();
    console.log('🔥 Admin: Firestore data loaded');
  } catch(e) { console.warn('Firebase init load failed:', e); }
}
fbInit();
